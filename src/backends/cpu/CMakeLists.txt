# CPU backend (AVX2). Standalone-capable: `cmake -S src/backends/cpu -B build-cpu` needs only a C++20 compiler
# (no CUDA). Also pulled into the main build via add_subdirectory (option QV_CPU).
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  cmake_minimum_required(VERSION 3.20)
  project(q_volta_cpu LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 20)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
  endif()
  enable_testing()
endif()
find_package(Threads REQUIRED)
get_filename_component(QV_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "arm64|aarch64|ARM64")
  set(QV_CPU_ARM ON)
  if(APPLE)
    set(QV_CPU_FLAGS -O3 -mcpu=native)
  elseif(ANDROID)
    # Snapdragon 8 Gen 2 (X3/A715/A710/A510): ARMv9.0-A = dotprod, fp16, i8mm, bf16
    set(QV_CPU_FLAGS -O3 -march=armv8.6-a+dotprod+fp16+i8mm+bf16)
  else()
    set(QV_CPU_FLAGS -O3 -march=armv8.2-a+dotprod+fp16)
  endif()
  set(QV_GEMV_SRC cpu_gemv_neon.cpp)
else()
  set(QV_CPU_ARM OFF)
  set(QV_CPU_FLAGS -O3 -mavx2 -mfma -mf16c -mbmi2)
  set(QV_GEMV_SRC cpu_gemv.cpp)
endif()

# quantized GEMV + thread pool
add_library(qv_cpu STATIC ${QV_GEMV_SRC})
target_include_directories(qv_cpu PUBLIC ${CMAKE_CURRENT_LIST_DIR})
target_compile_options(qv_cpu PUBLIC ${QV_CPU_FLAGS})
target_link_libraries(qv_cpu PUBLIC Threads::Threads)

# layer ops (GDN, conv, norms, rope, attention, embed, argmax)
add_library(qv_cpu_ops STATIC ops.cpp ink_ops.cpp)
target_include_directories(qv_cpu_ops PUBLIC ${CMAKE_CURRENT_LIST_DIR})
target_compile_options(qv_cpu_ops PUBLIC ${QV_CPU_FLAGS})

add_executable(test_cpu_gemv ${QV_ROOT}/tests/test_cpu_gemv.cpp)
target_link_libraries(test_cpu_gemv PRIVATE qv_cpu)
add_test(NAME cpu_gemv COMMAND test_cpu_gemv ${QV_ROOT}/tests/data/pack_fixture.txt)

add_executable(test_cpu_gemm ${QV_ROOT}/tests/test_cpu_gemm.cpp)
target_link_libraries(test_cpu_gemm PRIVATE qv_cpu)
add_test(NAME cpu_gemm COMMAND test_cpu_gemm)

add_executable(test_cpu_gemv_multi ${QV_ROOT}/tests/test_cpu_gemv_multi.cpp)
target_link_libraries(test_cpu_gemv_multi PRIVATE qv_cpu)
add_test(NAME cpu_gemv_multi COMMAND test_cpu_gemv_multi)

add_executable(test_cpu_ops ${QV_ROOT}/tests/test_cpu_ops.cpp)
target_link_libraries(test_cpu_ops PRIVATE qv_cpu_ops)
add_test(NAME cpu_ops COMMAND test_cpu_ops)

add_executable(test_cpu_ink_ops ${QV_ROOT}/tests/test_cpu_ink_ops.cpp)
target_link_libraries(test_cpu_ink_ops PRIVATE qv_cpu_ops)
add_test(NAME cpu_ink_ops COMMAND test_cpu_ink_ops)

if(NOT QV_CPU_ARM)  # x86-specific benches (TSC, AVX2 stream ceiling); ARM uses bench_neon_gemv
  add_executable(bench_cpu_gemv ${QV_ROOT}/benchmarks/bench_cpu_gemv.cpp)
  target_link_libraries(bench_cpu_gemv PRIVATE qv_cpu)
  add_executable(bench_cpu_ops ${QV_ROOT}/benchmarks/bench_cpu_ops.cpp)
  target_link_libraries(bench_cpu_ops PRIVATE qv_cpu_ops Threads::Threads)
endif()

# CPU engine (src/runtime/cpu_engine.cpp) + CUDA-free driver
if(NOT TARGET qv_core)
  add_library(qv_core STATIC ${QV_ROOT}/src/model/model_file.cpp)
  target_include_directories(qv_core PUBLIC ${QV_ROOT}/include)
endif()
add_library(qv_cpu_engine STATIC ${QV_ROOT}/src/runtime/cpu_engine.cpp ${QV_ROOT}/src/runtime/cpu_inkling_engine.cpp)
target_link_libraries(qv_cpu_engine PUBLIC qv_core qv_cpu qv_cpu_ops)
option(QV_CPU_PROF "per-phase timers in the CPU engine" OFF)
if(QV_CPU_PROF)
  target_compile_definitions(qv_cpu_engine PRIVATE QV_CPU_PROF)
endif()
target_include_directories(qv_cpu_engine PUBLIC ${QV_ROOT}/src/runtime)
add_executable(qv_ink_cpu ${QV_ROOT}/tools/qv_ink_cpu.cpp)
target_link_libraries(qv_ink_cpu PRIVATE qv_cpu_engine)
add_executable(test_cpu_mtp ${QV_ROOT}/tests/test_cpu_mtp.cpp)
target_link_libraries(test_cpu_mtp PRIVATE qv_cpu_engine)
# needs the model (E0 .qv.bin/.qv.json); skips itself when the prefix is missing. Override: -DQV_MTP_TEST_MODEL=prefix
set(QV_MTP_TEST_MODEL "/workspace/q-volta-models/qv/qwen38-27b-e0" CACHE STRING "model prefix for test_cpu_mtp")
add_test(NAME cpu_mtp COMMAND test_cpu_mtp ${QV_MTP_TEST_MODEL} 64 32)
if(NOT TARGET qv_run_cpu)
  add_executable(qv_run_cpu ${QV_ROOT}/tools/qv_run.cpp)
  target_link_libraries(qv_run_cpu PRIVATE qv_cpu_engine)
endif()

  add_executable(bench_neon_gemv ${QV_ROOT}/benchmarks/bench_neon_gemv.cpp)
  target_link_libraries(bench_neon_gemv PRIVATE qv_cpu)
